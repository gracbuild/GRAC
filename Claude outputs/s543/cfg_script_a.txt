<script>
(function () {
    'use strict';

    const root = document.getElementById('oaCfgRoot');
    const wf   = window.__wfCommon;
    if (!root || !wf) return;

    const API = '/practice/api/org-assurance';
    const U   = p => String(window.appBasePath || window.pmPathBase || '').replace(/\/+$/, '') + p;
    const esc = wf.escape;
    const pick = function (o, a, b) { return o ? (o[a] != null ? o[a] : o[b]) : null; };

    const state = {
        mode: 'list', orgId: '', definitionId: '', name: '', statusCode: '',
        steps: [], rows: {},
        search: '', page: 1, pageSize: 25, total: 0
    };

    const el = {
        heading  : document.getElementById('oaCfgHeading'),
        subtitle : document.getElementById('oaCfgSubtitle'),
        actions  : document.getElementById('oaCfgActionsBtn'),
        back     : document.getElementById('oaCfgBack'),
        backText : document.getElementById('oaCfgBackLabel'),
        body     : document.getElementById('oaCfgBody'),
        org      : document.getElementById('oaCfgFilterOrganization'),
        submit   : document.getElementById('oaCfgSubmit'),
        approve  : document.getElementById('oaCfgApprove'),
        dlg      : document.getElementById('oaCfgStepDialog')
    };

    function chip(status) {
        const shell = window.__oaFlowShell;
        return shell ? shell.chipHtml(status || 'NotConfigured') : esc(status || '');
    }
    function box(message, type) {
        if (window.gracUi) return window.gracUi.alert(message, { type: type });
        window.alert(message);
        return Promise.resolve();
    }

    // -----------------------------------------------------------------
    // Mode
    // -----------------------------------------------------------------
    function setMode(mode) {
        state.mode = mode;
        root.setAttribute('data-mode', mode);
        el.actions.hidden = mode !== 'view';
        el.back.hidden = mode === 'list';
        el.backText.textContent = 'Back to audit list';
        if (mode === 'list') {
            el.heading.textContent  = 'Audit Configuration';
            el.subtitle.textContent = 'Draft and Under Review audits: scope, question sets, workflow, scoring, trigger and scope resolution.';
        }
        window.scrollTo(0, 0);
    }

    el.back.addEventListener('click', function () {
        state.definitionId = '';
        state.statusCode = '';
        setMode('list');
        reload();
    });

    // -----------------------------------------------------------------
    // Review (542, folded into 543): Submit for Review / Approve.
    // Same endpoints and body the Audit Definition menu used
    // (POST definitions/{id}/submit | approve). The 537 gate still checks
    // the configuration on the server (THROW 53612).
    // -----------------------------------------------------------------
    function reviewItems(id, statusCode, openSteps) {
        const items = [];
        if (statusCode === 'Draft')
            items.push({ icon: 'fa-paper-plane', label: 'Submit for Review', run: function () { submitForReview(id, openSteps()); } });
        if (statusCode === 'UnderReview')
            items.push({ icon: 'fa-check', label: 'Approve', run: function () { approve(id); } });
        return items;
    }

    async function submitForReview(id, open) {
        if (open && open.length) {
            await box('Complete every step before submitting the audit for review. Not complete: ' + open.join(', ') + '.', 'warning');
            return;
        }
        const reason = await window.gracUi.prompt('Reason (optional):',
            { title: 'Submit for Review', inputLabel: 'Reason', confirmText: 'Submit' });
        if (reason === null) return;   // cancelled
        await transition(id, 'submit', reason, 'Submitted for review.');
    }
    async function approve(id) {
        const reason = await window.gracUi.prompt('Reason (optional):',
            { title: 'Approve', inputLabel: 'Reason', confirmText: 'Approve' });
        if (reason === null) return;
        await transition(id, 'approve', reason, 'Audit approved.');
    }
    async function transition(id, action, reason, done) {
        try {
            const r = await fetch(U(API + '/definitions/' + encodeURIComponent(id) + '/' + action), {
                method      : 'POST',
                credentials : 'same-origin',
                headers     : { 'Content-Type': 'application/json' },
                body        : JSON.stringify({
                    organizationId : Number(state.orgId),
                    definitionId   : Number(id),
                    reasonText     : reason || null,
                    actor          : null
                })
            });
            const j = await r.json().catch(function () { return null; });
            if (!r.ok) { await box('Failed: ' + ((j && (j.error || j.Error)) || ('HTTP ' + r.status)), 'error'); return; }
            await box(done, 'success');
        } catch (err) {
            await box('Failed: ' + (err.message || err), 'error');
            return;
        }
        if (state.mode === 'view' && state.definitionId === String(id)) loadView(); else reload();
    }

    if (window.gracPageActions) {
        window.gracPageActions.attach('oaCfgActionsBtn', function () {
            return state.definitionId ? reviewItems(state.definitionId, state.statusCode, openStepNames) : [];
        });
    }
    el.submit.addEventListener('click', function () { submitForReview(state.definitionId, openStepNames()); });
    el.approve.addEventListener('click', function () { approve(state.definitionId); });

    // -----------------------------------------------------------------
    // List
    // -----------------------------------------------------------------
    const LIST_STEPS = [
        ['Scope Builder',    'scopeStatus',           'ScopeStatus'],
        ['Question Sets',    'questionsStatus',       'QuestionsStatus'],
        ['Workflow',         'workflowStatus',        'WorkflowStatus'],
        ['Scoring',          'scoringStatus',         'ScoringStatus'],
        ['Trigger',          'triggerStatus',         'TriggerStatus'],
        ['Scope Resolution', 'scopeResolutionStatus', 'ScopeResolutionStatus']
    ];

    async function reload() {
        if (!state.orgId) {
            el.body.innerHTML = '<tr><td colspan="13" class="pm-empty">Select an organization to load audits.</td></tr>';
            wf.updatePager('oa-cfg', { pageNumber: 1, totalCount: 0, pageSize: 1 });
            return;
        }
        el.body.innerHTML = '<tr><td colspan="13" class="pm-empty">Loading...</td></tr>';
        try {
            const qs = new URLSearchParams();
            qs.set('organizationId', state.orgId);
            if (state.search) qs.set('search', state.search);
            qs.set('page', String(state.page));
            qs.set('pageSize', String(state.pageSize));
            const r = await fetch(U(API + '/configuration-audits?' + qs.toString()), { credentials: 'same-origin' });
            const body = await r.json().catch(function () { return null; });
            if (!r.ok) throw new Error((body && (body.error || body.Error)) || ('HTTP ' + r.status));
            state.total = Number(pick(body, 'totalCount', 'TotalCount') || 0);
            const rows = pick(body, 'rows', 'Rows') || [];
            renderRows(rows);
            wf.updatePager('oa-cfg', { pageNumber: state.page, totalCount: state.total, pageSize: state.pageSize, rowCount: rows.length });
        } catch (err) {
            el.body.innerHTML = '<tr><td colspan="13" class="pm-empty">Failed to load: ' + esc(err.message || err) + '</td></tr>';
        }
    }

    function renderRows(rows) {
        state.rows = {};
        if (!rows.length) {
            el.body.innerHTML = '<tr><td colspan="13" class="pm-empty">No audits to configure. A new audit appears here once its details are saved in Audit Definition.</td></tr>';
            return;
        }
        el.body.innerHTML = rows.map(function (row) {
            const p = function (a, b) { return pick(row, a, b); };
            const id = String(p('definitionId', 'DefinitionId'));
            state.rows[id] = row;
            const owner = wf.personWithRole(p('ownerDisplayName', 'OwnerDisplayName'), p('ownerRoleName', 'OwnerRoleName'));
            const ver = p('currentVersionLabel', 'CurrentVersionLabel') || (p('currentVersionNumber', 'CurrentVersionNumber') ? String(p('currentVersionNumber', 'CurrentVersionNumber')) : '');
            const done = Number(p('configuredSteps', 'ConfiguredSteps') || 0);
            const all  = Number(p('configurationSteps', 'ConfigurationSteps') || 6);
            return '<tr class="pm-row-clickable" data-id="' + esc(id) + '">'
                + '<td>' + esc(p('definitionCode', 'DefinitionCode')) + '</td>'
                + '<td>' + esc(p('definitionName', 'DefinitionName')) + '</td>'
                + '<td>' + esc(owner) + '</td>'
                + '<td>' + esc(ver) + '</td>'
                + '<td>' + esc(p('currentStatusName', 'CurrentStatusName')) + '</td>'
                + LIST_STEPS.map(function (s) { return '<td>' + chip(p(s[1], s[2])) + '</td>'; }).join('')
                + '<td>' + esc(done + ' of ' + all) + '</td>'
                + '<td><button type="button" class="pm-action-trigger" data-id="' + esc(id) + '"'
                + ' aria-haspopup="menu" aria-expanded="false" title="Actions">'
                + '<i class="fas fa-ellipsis-v fa-solid fa-ellipsis-vertical" aria-hidden="true"></i>'
                + '<span class="visually-hidden">Actions</span></button></td>'
                + '</tr>';
        }).join('');

        // 3-dot: View, then Submit for Review / Approve by status.
        if (window.gracPageActions) {
            el.body.querySelectorAll('.pm-action-trigger').forEach(function (btn) {
                const id  = btn.getAttribute('data-id');
                const row = state.rows[id];
                window.gracPageActions.attach(btn, function () {
                    return [{ icon: 'fa-up-right-from-square', label: 'View', run: function () { openView(id); } }]
                        .concat(reviewItems(id, pick(row, 'currentStatusCode', 'CurrentStatusCode') || '', function () {
                            return LIST_STEPS.filter(function (s) { return pick(row, s[1], s[2]) !== 'Completed'; })
                                             .map(function (s) { return s[0]; });
                        }));
                });
            });
        }
    }

    // Row click = View (the Audit Definition list's pattern).
    el.body.addEventListener('click', function (e) {
        if (e.target.closest('.pm-action-trigger, a, button, input, label, select')) return;
        const tr = e.target.closest('tr.pm-row-clickable[data-id]');
        if (tr) openView(tr.getAttribute('data-id'));
    });
